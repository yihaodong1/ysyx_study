### 3
```
#!/usr/bin/env bash
count=0
touch log.txt
while :;
do
        ./test.sh >> log.txt 2>&1
        [[ $? -eq 0 ]] || break
        let count+=1;
done
cat log.txt
echo "successfully run $count times"
```
